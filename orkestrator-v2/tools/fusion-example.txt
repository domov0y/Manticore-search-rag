function multi_model_analysis_with_jude($history, $tools, $params, $model_list)
{ 
  $tmp_result=array();
  foreach ($model_list as $model)
  {
    
	$myparams=$params;   
	$myparams['modelname']=$model;   
    $tmp_result[$model]=llm_conversation(history,tools,myparams);
  }

  $analyse=array();
$system_prompt="Ты аналитик.

Тебе дан исходный вопрос пользователя и ответы нескольких языковых моделей.

Не отвечай на вопрос пользователя.
Не переписывай ответы моделей.
Не пытайся составить лучший ответ.

Твоя задача — сравнить ответы и подготовить аналитическую сводку.

Выведи результат в следующем формате.

CONSENSUS
Факты и утверждения, с которыми согласны большинство моделей.

DIFFERENCES
Основные различия между ответами.

UNIQUE
Полезные идеи, которые встретились только в отдельных ответах.

ERRORS
Замеченные логические ошибки, противоречия, фактические ошибки или сомнительные утверждения.

MISSING
Какие важные аспекты вопроса не были раскрыты ни одной моделью.

CONFIDENCE
Оцени по шкале от 0 до 100, насколько ответы позволяют уверенно ответить пользователю.

Не придумывай новых фактов.
Используй только информацию, содержащуюся в ответах моделей.
Если невозможно определить, какая модель права, так и напиши." ;

  $analyse[]=['role'=>'system' , 'content'=>$system_prompt];

  $text = "ВОПРОС ПОЛЬЗОВАТЕЛЯ\n\n";
  $text .= end($history)['content'];
  $text .= "\n\n";

  foreach ($tmp_result as $model=>$answer)
  {
      $text.="=====================\n";
      $text.="MODEL: ".$model."\n";
      $text.="=====================\n";
      $text.=$answer['content']."\n\n";
  }

  $analyse[]=[
     'role'=>'user',
     'content'=>$text
  ];
  
  return llm_conversation($analyse,[],$params);
   
}
